  <script>
    const out = document.getElementById('out');
    const show = data => out.textContent = typeof data === 'string' ? data : JSON.stringify(data, null, 2);
    async function catalog() {
      const q = document.getElementById('catalog').value;
      const r = await fetch('/api/catalog?name=' + encodeURIComponent(q));
      show(await r.json());
    }
    async function person() {
      const q = document.getElementById('person').value;
      const r = await fetch('/api/collaborators?name=' + encodeURIComponent(q));
      show(await r.json());
    }
    async function ticket() {
      const q = document.getElementById('ticket').value;
      const r = await fetch('/api/tickets?protocol=' + encodeURIComponent(q));
      show(await r.json());
    }
    async function login() {
      const username = document.getElementById('username').value;
      const password = document.getElementById('password').value;
      const r = await fetch('/api/session', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ username, password })
      });
      show(await r.json());
    }
